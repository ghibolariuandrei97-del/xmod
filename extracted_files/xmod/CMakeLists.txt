cmake_minimum_required(VERSION 3.10)
project(xmod C CXX)
# Linux  (32-bit):   cmake -B build -DCMAKE_BUILD_TYPE=Release && cmake --build build -j       -> xmod_mm_i386.so
# Windows (MinGW cross from Linux): cmake -B build-win -DCMAKE_TOOLCHAIN_FILE=cmake/mingw-i686.cmake -DCMAKE_BUILD_TYPE=Release && cmake --build build-win -j   -> xmod_mm.dll
# Windows (MSVC, x86!):   cmake -B build-msvc -A Win32 && cmake --build build-msvc --config Release   -> xmod_mm.dll
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 99)
set(LUA ${CMAKE_SOURCE_DIR}/deps/lua)
file(GLOB LUA_SRC ${LUA}/*.c)
list(FILTER LUA_SRC EXCLUDE REGEX "(lua|luac|onelua|ltests)\\.c$")
set(HL ${CMAKE_SOURCE_DIR}/deps/hlsdk)
set(SRC src/main.cpp src/core.cpp src/api_util.cpp src/api_player.cpp src/api_ent.cpp src/api_msg.cpp src/api_cmd.cpp
        src/api_misc.cpp src/reapi.cpp src/ham.cpp src/hooks_api.cpp src/ext_api.cpp src/diag.cpp src/vault.cpp src/nexus_server.cpp)
if(MSVC)
  add_library(xmod_mm SHARED ${SRC} ${LUA_SRC} src/xmod_mm.def)
  target_compile_options(xmod_mm PRIVATE /O2 /EHsc /MT /wd4996 /wd4244 /wd4267 /wd4305 /D_CRT_SECURE_NO_WARNINGS)
  set_target_properties(xmod_mm PROPERTIES PREFIX "")
else()
  set(CMAKE_C_FLAGS   "${CMAKE_C_FLAGS} -O2 -fPIC")
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O2 -fPIC -fno-exceptions -fno-rtti -pthread -Wno-write-strings -Wno-endif-labels -Wno-attributes -fpermissive -D_WIN32_WINNT=0x0601")
  if(WIN32)   # MinGW
    add_library(xmod_mm SHARED ${SRC} ${LUA_SRC})
    set_target_properties(xmod_mm PROPERTIES PREFIX "" LINK_FLAGS "-static -static-libgcc -static-libstdc++ -Wl,--kill-at")
  else()      # Linux, 32-bit (GoldSrc is 32-bit)
    set(CMAKE_C_FLAGS   "${CMAKE_C_FLAGS} -m32 -DLUA_USE_POSIX")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -m32 -fvisibility=hidden")
    add_library(xmod_mm SHARED ${SRC} ${LUA_SRC})
    set_target_properties(xmod_mm PROPERTIES OUTPUT_NAME "xmod_mm_i386" PREFIX "" LINK_FLAGS "-m32 -static-libgcc -static-libstdc++ -Wl,--no-undefined")
    target_link_libraries(xmod_mm dl m pthread)
  endif()
endif()
target_include_directories(xmod_mm PRIVATE ${LUA} ${HL}/dlls ${HL}/engine ${HL}/common ${HL}/pm_shared ${HL}/public src include)

# ---------------------------------------------------------------------------------------------- extensions (Metamod plugins that plug into xmod)
option(XMOD_BUILD_EXTENSIONS "Build the bundled xmod extensions" ON)
function(xmod_extension NAME)      # xmod_extension(xmod_foo src1.cpp src2.cpp ...)  ->  <NAME>_mm_i386.so / <NAME>_mm.dll
  add_library(${NAME} SHARED ${ARGN})
  if(WIN32)
    set_target_properties(${NAME} PROPERTIES PREFIX "" OUTPUT_NAME "${NAME}_mm" LINK_FLAGS "-static -static-libgcc -static-libstdc++ -Wl,--kill-at")
  else()
    set_target_properties(${NAME} PROPERTIES PREFIX "" OUTPUT_NAME "${NAME}_mm_i386" LINK_FLAGS "-m32 -static-libgcc -static-libstdc++ -Wl,--no-undefined")
    target_link_libraries(${NAME} dl m pthread)
  endif()
  target_include_directories(${NAME} PRIVATE ${HL}/dlls ${HL}/engine ${HL}/common ${HL}/pm_shared ${HL}/public ${CMAKE_SOURCE_DIR}/include)
endfunction()
if(XMOD_BUILD_EXTENSIONS AND NOT MSVC)
  file(GLOB EXT_DIRS LIST_DIRECTORIES true ${CMAKE_SOURCE_DIR}/extensions/*)
  foreach(d ${EXT_DIRS})
    if(IS_DIRECTORY ${d} AND EXISTS ${d}/CMakeLists.txt)
      add_subdirectory(${d})
    endif()
  endforeach()
endif()
